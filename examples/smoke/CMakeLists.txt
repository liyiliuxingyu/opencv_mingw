cmake_minimum_required(VERSION 3.20)
project(OpenCVMinGWSmoke LANGUAGES CXX)

if(NOT WIN32 OR NOT MINGW)
  message(FATAL_ERROR "This example validates the Windows MinGW SDK. Select its bundled g++.exe.")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This SDK requires an x86-64 MinGW compiler (64-bit pointers).")
endif()

find_package(OpenCV CONFIG REQUIRED COMPONENTS core imgproc imgcodecs videoio)

if(NOT CMAKE_BUILD_TYPE MATCHES "^(Release|Debug)$")
  message(FATAL_ERROR "Select the SDK configuration with -DCMAKE_BUILD_TYPE=Release or Debug.")
endif()
if(DEFINED OPENCV_EXPECTED_CONFIGURATION AND
   NOT CMAKE_BUILD_TYPE STREQUAL OPENCV_EXPECTED_CONFIGURATION)
  message(FATAL_ERROR "The consumer configuration must match the selected SDK.")
endif()
string(TOUPPER "${CMAKE_BUILD_TYPE}" _opencv_configuration)
if(CMAKE_BUILD_TYPE STREQUAL "Debug" AND TARGET opencv_world)
  message(FATAL_ERROR "The standalone Debug SDK must provide independent module DLLs.")
endif()
set(_opencv_link_report "configuration=${CMAKE_BUILD_TYPE}\n")
foreach(_opencv_module core imgproc imgcodecs videoio)
  set(_opencv_target "opencv_${_opencv_module}")
  if(NOT TARGET ${_opencv_target})
    message(FATAL_ERROR "OpenCV did not export ${_opencv_target}.")
  endif()
  # World builds expose component aliases. Inspect the actual shared library
  # so the check also works if an upstream alias becomes an interface target.
  set(_opencv_shared_target "${_opencv_target}")
  if(TARGET opencv_world)
    set(_opencv_shared_target opencv_world)
  endif()
  get_target_property(_opencv_configs ${_opencv_shared_target} IMPORTED_CONFIGURATIONS)
  if(NOT _opencv_configuration IN_LIST _opencv_configs)
    message(FATAL_ERROR "${_opencv_target} does not provide ${CMAKE_BUILD_TYPE}; configuration fallback is not accepted.")
  endif()
  get_target_property(_opencv_dll ${_opencv_shared_target} "IMPORTED_LOCATION_${_opencv_configuration}")
  get_target_property(_opencv_import_library ${_opencv_shared_target} "IMPORTED_IMPLIB_${_opencv_configuration}")
  if(NOT EXISTS "${_opencv_dll}" OR NOT EXISTS "${_opencv_import_library}")
    message(FATAL_ERROR "${_opencv_target} is missing its ${CMAKE_BUILD_TYPE} DLL or import library.")
  endif()
  if(CMAKE_BUILD_TYPE STREQUAL "Debug" AND NOT _opencv_dll MATCHES "d\\.dll$")
    message(FATAL_ERROR "${_opencv_target} did not select a Debug DLL.")
  endif()
  string(APPEND _opencv_link_report "${_opencv_target}=$<TARGET_FILE:${_opencv_shared_target}>\n")
endforeach()
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/opencv-link-selection.txt" CONTENT "${_opencv_link_report}")

add_executable(opencv_mingw_smoke main.cpp)
target_compile_features(opencv_mingw_smoke PRIVATE cxx_std_17)
target_compile_definitions(opencv_mingw_smoke PRIVATE OPENCV_SMOKE_CONFIGURATION="${CMAKE_BUILD_TYPE}")
set_target_properties(opencv_mingw_smoke PROPERTIES CXX_EXTENSIONS OFF)
target_include_directories(opencv_mingw_smoke PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(opencv_mingw_smoke PRIVATE ${OpenCV_LIBS})

enable_testing()
add_test(NAME opencv-sdk-smoke
  COMMAND opencv_mingw_smoke "${CMAKE_CURRENT_BINARY_DIR}/test-output")
set_tests_properties(opencv-sdk-smoke PROPERTIES TIMEOUT 120)
