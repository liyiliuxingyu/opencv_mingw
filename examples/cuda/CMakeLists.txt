cmake_minimum_required(VERSION 3.24)
project(OpenCVCudaSDKSmoke LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Use the x64 Visual Studio C++ compiler for this CUDA SDK.")
endif()
find_package(OpenCV CONFIG REQUIRED COMPONENTS core imgproc imgcodecs videoio cudaarithm cudaimgproc)
foreach(name opencv_cuda_smoke opencv_cuda_device)
  if(name STREQUAL "opencv_cuda_smoke")
    add_executable(${name} ../smoke/main.cpp)
  else()
    add_executable(${name} device.cpp)
  endif()
  target_compile_features(${name} PRIVATE cxx_std_17)
  target_compile_definitions(${name} PRIVATE OPENCV_SMOKE_CONFIGURATION="$<CONFIG>")
  target_include_directories(${name} PRIVATE ${OpenCV_INCLUDE_DIRS})
  target_link_libraries(${name} PRIVATE ${OpenCV_LIBS})
endforeach()
